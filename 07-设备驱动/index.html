7.2 字符设备驱动完整模板

下面是一个可编译运行的最小字符设备驱动，实现 /dev/hello 读写：

/* hello_drv.c */
#include <linux/module.h>
#include <linux/fs.h>
#include <linux/cdev.h>
#include <linux/device.h>
#include <linux/uaccess.h>

static dev_t devno;
static struct cdev hello_cdev;
static struct class *hello_class;
static char buffer[128] = "Hello from kernel!\n";
static ssize_t buf_len = 19;

static ssize_t hello_read(struct file *f, char __user *u, size_t n, loff_t *off) {
    if (*off >= buf_len) return 0;          // EOF
    n = min(n, (size_t)(buf_len - *off));
    if (copy_to_user(u, buffer + *off, n)) return -EFAULT;
    *off += n;
    return n;
}

static ssize_t hello_write(struct file *f, const char __user *u, size_t n, loff_t *off) {
    if (n > 127) n = 127;
    if (copy_from_user(buffer, u, n)) return -EFAULT;
    buffer[n] = '\0';
    buf_len = n;
    return n;
}

static const struct file_operations hello_fops = {
    .owner = THIS_MODULE,
    .read  = hello_read,
    .write = hello_write,
};

static int __init hello_init(void) {
    alloc_chrdev_region(&devno, 0, 1, "hello");
    cdev_init(&hello_cdev, &hello_fops);
    cdev_add(&hello_cdev, devno, 1);
    hello_class = class_create("hello_class");
    device_create(hello_class, NULL, devno, NULL, "hello");  // → /dev/hello
    pr_info("hello driver loaded\n");
    return 0;
}

static void __exit hello_exit(void) {
    device_destroy(hello_class, devno);
    class_destroy(hello_class);
    cdev_del(&hello_cdev);
    unregister_chrdev_region(devno, 1);
}

module_init(hello_init);
module_exit(hello_exit);
MODULE_LICENSE("GPL");

# Makefile
obj-m += hello_drv.o
all:
	make -C /lib/modules/$(shell uname -r)/build M=$(PWD) modules
clean:
	make -C /lib/modules/$(shell uname -r)/build M=$(PWD) clean

# 测试
$ make
$ sudo insmod hello_drv.ko
$ cat /dev/hello              # Hello from kernel!
$ echo "hi there" > /dev/hello
$ cat /dev/hello              # hi there
$ sudo rmmod hello_drv

7.3 平台设备 + 设备树 (ARM 嵌入式必懂)

x86 设备通过 ACPI 自动发现；ARM 等嵌入式没有 ACPI，靠设备树 (DTS) 描述硬件：

/* my_board.dts */
&i2c1 {
    eeprom@50 {
        compatible = "atmel,24c64";     // 关键: 用于匹配驱动
        reg = <0x50>;                    // I2C 地址
        pagesize = <32>;
    };
};

/* 驱动 (drivers/misc/eeprom/at24.c 简化) */
static const struct of_device_id at24_of_match[] = {
    { .compatible = "atmel,24c64" },
    { },
};

static struct platform_driver at24_driver = {
    .driver = {
        .name = "at24",
        .of_match_table = at24_of_match,
    },
    .probe = at24_probe,
    .remove = at24_remove,
};
module_platform_driver(at24_driver);

7.4 中断处理 — 上下半部

/* 上半部 (top half) — 中断上下文，必须快! */
static irqreturn_t my_irq_handler(int irq, void *dev_id) {
    // 1. 读寄存器确认是自己的中断
    u32 status = readl(dev->regs + IRQ_STATUS);
    if (!status) return IRQ_NONE;

    // 2. 清中断、关本中断
    writel(status, dev->regs + IRQ_CLEAR);

    // 3. 紧急工作做完就走 — 复杂处理交给下半部
    schedule_work(&dev->bh_work);

    return IRQ_HANDLED;
}

/* 下半部 (bottom half) — 进程上下文，可 sleep */
static void my_bh_work(struct work_struct *w) {
    struct my_dev *dev = container_of(w, struct my_dev, bh_work);
    // 慢速操作: 读 DMA 数据、唤醒等待进程...
    wake_up_interruptible(&dev->wait);
}

/* 注册中断 */
request_irq(irq, my_irq_handler, IRQF_SHARED, "my-dev", dev);

下半部的三种方式

机制上下文能 sleep?使用场景
softirq软中断上下文否网络收发、定时器 (固定 10 种)
tasklet软中断上下文否逐渐被 workqueue 取代
workqueue进程上下文是需要睡眠/分配的延后处理（首选）
threaded IRQ专用内核线程是低延迟 + 需睡眠 (RT 内核必备)

7.5 DMA 简述

/* DMA 一致性内存分配 — 内核 + 设备都能访问 */
dma_addr_t dma_handle;
void *cpu_addr = dma_alloc_coherent(dev, size, &dma_handle, GFP_KERNEL);

/* 流式 DMA — 单向，更省事但要 sync */
dma_handle = dma_map_single(dev, buffer, len, DMA_TO_DEVICE);
// 把 dma_handle 给硬件 → 硬件 DMA 读完
dma_unmap_single(dev, dma_handle, len, DMA_TO_DEVICE);
