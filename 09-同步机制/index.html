09同步机制 — 自旋锁、Mutex、RCU、Memory Barriers

SMP 时代，"并发即正义，并发即灾难"。本章覆盖从原子操作到 RCU 的完整同步谱系，并教你选锁。

9.1 为什么需要同步

三种"看起来好但其实坏"的代码
  1. 没保护的全局变量：counter++ 实际是 read-modify-write 三步，多核同时执行结果错乱
  2. 错误的双重检查：if (!ptr) ptr = alloc(); 多核可能 alloc 多次
  3. 遗漏 barrier：编译器 / CPU 乱序 → 用户看到"不可能"的状态

9.2 原子操作 — 基石

#include <linux/atomic.h>

atomic_t v = ATOMIC_INIT(0);

atomic_inc(&v);              // v++
atomic_dec(&v);              // v--
atomic_add(5, &v);          // v += 5
atomic_read(&v);             // 读取 (READ_ONCE)
atomic_set(&v, 0);          // 写入 (WRITE_ONCE)

// 返回旧值/新值的版本
int old = atomic_xchg(&v, 10);
int old = atomic_cmpxchg(&v, expect, 10);

// 检查零 (常用于引用计数)
if (atomic_dec_and_test(&v)) {
    // v 减 1 后等于 0 → 安全释放
    kfree(obj);
}

原子操作在 x86 上对应 LOCK 前缀指令（如 lock incl），保证 cache line 独占。

9.3 自旋锁 (spinlock) — 短临界区

DEFINE_SPINLOCK(my_lock);

/* 不能在持锁时 sleep / 调度 — 中断/软中断仍可发生 */
spin_lock(&my_lock);
... critical section ...
spin_unlock(&my_lock);

/* 中断处理也用此锁? 必须用 _irqsave 版本 */
unsigned long flags;
spin_lock_irqsave(&my_lock, flags);
...
spin_unlock_irqrestore(&my_lock, flags);
为什么自旋锁的临界区不能 sleep

持自旋锁时，本 CPU 禁用抢占。若 sleep → 另一进程跑上来 → 也试图拿锁 → 等不到永远不让出 → 死锁。 所以自旋锁的代码内：不能调 schedule()，不能 kmalloc(GFP_KERNEL)（可能回收 → 可能 sleep），不能 copy_to_user（可能缺页 → 可能 sleep）。

9.4 mutex / semaphore — 长临界区

DEFINE_MUTEX(big_lock);

mutex_lock(&big_lock);       // 拿不到 → 睡眠
mutex_lock_interruptible(&big_lock);  // 可被信号中断
mutex_trylock(&big_lock);    // 立刻返回，不睡
... critical section (允许 sleep!) ...
mutex_unlock(&big_lock);

9.5 读写锁与 seqlock

/* rwlock — 多读单写 (适合读多写少) */
DEFINE_RWLOCK(my_rwlock);
read_lock(&my_rwlock);   ... read_unlock(&my_rwlock);
write_lock(&my_rwlock);  ... write_unlock(&my_rwlock);

/* seqlock — 写优先，读可重试 (jiffies/xtime 用这个) */
seqlock_t timer_lock;
unsigned seq;
do {
    seq = read_seqbegin(&timer_lock);
    ... 读多个字段 ...
} while (read_seqretry(&timer_lock, seq));  // 写者来过 → 重读

9.6 RCU — Read-Copy-Update（内核同步皇冠）

RCU 一句话原理

读者完全无锁、零开销。写者不修改共享数据，而是复制一份、修改副本、原子地替换指针。 旧版本数据等所有"当时在读"的进程都退出（grace period）后再释放。

/* 读 — 极度便宜 */
rcu_read_lock();      // 只是禁用抢占,几乎零开销
struct foo *p = rcu_dereference(global_ptr);
if (p) use(p->data);
rcu_read_unlock();

/* 写 — 复制-修改-替换 */
struct foo *new = kmalloc(sizeof(*new), GFP_KERNEL);
*new = *old_ptr;
new->data = 42;
rcu_assign_pointer(global_ptr, new);   // 原子替换
synchronize_rcu();    // 等所有读者退出 (老进程完成 read)
kfree(old);           // 安全释放

/* 或异步版本 (推荐用于热路径) */
call_rcu(&old->rcu_head, free_callback);

RCU 用在哪：路由表、cgroup 链表、dentry 缓存、内核模块列表、IDR、percpu 数据……几乎所有"读多写少"场景。

9.7 Per-CPU 变量 — 彻底避免共享

DEFINE_PER_CPU(long, my_counter);     // 每个 CPU 一份独立的 my_counter

/* 读写本 CPU 的变量 (需禁抢占) */
preempt_disable();
this_cpu_inc(my_counter);
preempt_enable();

/* 读所有 CPU 累加 */
long total = 0;
int cpu;
for_each_possible_cpu(cpu)
    total += per_cpu(my_counter, cpu);

9.8 内存屏障 (Memory Barriers)

/* 编译器屏障 — 仅禁止编译器重排 */
barrier();

/* CPU 屏障 (x86 上前两个是空操作，arm/ppc 上有真指令) */
smp_rmb();           // 读屏障
smp_wmb();           // 写屏障
smp_mb();            // 全屏障

/* 典型场景: 单生产者单消费者环形队列 */
// 生产者
ring[head] = data;
smp_wmb();           // 确保数据先写入,再更新 head
WRITE_ONCE(ring->head, head+1);

// 消费者
while (READ_ONCE(ring->tail) == READ_ONCE(ring->head)) continue;
smp_rmb();           // 确保读到 head 后再读数据
data = ring[tail];

9.9 futex — 用户态锁的内核基础

pthread_mutex 性能秘诀：无竞争时纯用户态原子操作，有竞争时才陷入内核 futex_wait。

/* 简化的伪代码 */
mutex_lock(m):
    if (atomic_cmpxchg(&m->state, 0, 1) == 0)
        return;            // 快路径: 纯用户态!
    futex(&m->state, FUTEX_WAIT, 1, ...);  // 慢路径: 进内核睡

mutex_unlock(m):
    if (atomic_xchg(&m->state, 0) == 1)
        return;
    futex(&m->state, FUTEX_WAKE, 1, ...);  // 唤醒一个

9.10 选锁速查表

场景推荐
临界区 < 100ns，不睡spinlock
会 sleep / 持锁久mutex
读极多写极少RCU
读多写少但写不能等rwlock / seqlock
计数 / 统计atomic_t 或 percpu
等待事件wait_queue + completion
多个锁需要按序mutex_lock_nested + lockdep

9.11 lockdep — 死锁检测器

CONFIG_PROVE_LOCKING 开启后，内核运行时记录每把锁的获取顺序。一旦发现潜在死锁（哪怕这次没真死），立刻 dmesg 大字报：

[ 1234.5] ======================================================
[ 1234.5] WARNING: possible recursive locking detected
[ 1234.5] ...
[ 1234.5] Possible unsafe locking scenario:
[ 1234.5]   CPU0
[ 1234.5]   ----
[ 1234.5]   lock(&inode->i_mutex);
[ 1234.5]   lock(&inode->i_mutex);
[ 1234.5] *** DEADLOCK ***
