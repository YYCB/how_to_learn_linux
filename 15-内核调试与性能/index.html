15.2 printk — 最简单也最常用

/* 日志级别 */
pr_emerg(...);    // KERN_EMERG  0  系统不可用
pr_alert(...);    // KERN_ALERT  1  立即处理
pr_crit(...);     // KERN_CRIT   2  严重错误
pr_err(...);      // KERN_ERR    3  错误
pr_warn(...);     // KERN_WARNING 4 警告
pr_notice(...);   // KERN_NOTICE 5  注意事项
pr_info(...);     // KERN_INFO   6  信息
pr_debug(...);    // KERN_DEBUG  7  调试

/* 限速版本 — 防止刷屏 */
pr_warn_ratelimited("oops happened %d times\n", count);

/* 一次性 — 只打印第一次 */
pr_warn_once("deprecated API\n");

dmesg --level=err,warn
sudo dmesg -wH                  # 实时跟踪，人类友好时间戳
echo 7 > /proc/sys/kernel/printk    # 显示所有级别

15.3 ftrace — 内核内建追踪框架

# 1. 跟踪所有函数调用 (function tracer)
cd /sys/kernel/debug/tracing
echo function > current_tracer
cat trace_pipe | head

# 2. 函数图 (function_graph) — 看调用关系
echo function_graph > current_tracer
echo sys_open > set_graph_function
cat trace_pipe

#  3) cpu | duration | function
#  0)               |  sys_open() {
#  0)               |    do_sys_open() {
#  0)   0.420 us    |      _raw_spin_lock();
#  0)   1.250 us    |      do_filp_open() {
#  0)   8.470 us    |    }
#  0)  12.100 us    |  }

# 3. 跟特定事件 (events)
echo 1 > events/sched/sched_switch/enable
echo 1 > events/syscalls/sys_enter_open/enable

# 4. 过滤特定进程
echo $(pidof myapp) > set_ftrace_pid

function_graph 的杀手锏

# 看哪个函数耗时最多 (按级排序输出)
echo function_graph > current_tracer
echo nop > set_ftrace_filter
echo nop > set_graph_function
echo schedule > set_graph_function     # 关注 schedule 子树
echo 10000 > tracing_thresh             # 只显示 >10us 的
cat trace_pipe

15.4 perf — 性能分析瑞士军刀

# 实时看哪些函数 CPU 占用最高 (top)
sudo perf top

# 录制 30 秒采样 (99Hz: 质数避免与周期性事件同步)
sudo perf record -F 99 -a -g -- sleep 30
sudo perf report                   # 交互式 TUI

# 输出火焰图 (Brendan Gregg)
git clone https://github.com/brendangregg/FlameGraph
sudo perf script | ./FlameGraph/stackcollapse-perf.pl | ./FlameGraph/flamegraph.pl > flame.svg

# 跟踪特定事件 (PMU)
sudo perf stat -e cycles,instructions,cache-misses,branch-misses -- ./my_app

# 调度延迟
sudo perf sched record -- sleep 5
sudo perf sched latency
sudo perf sched map

15.5 KASAN — 内存错误检测器

KASAN (Kernel Address Sanitizer) 用影子内存追踪每字节的访问权限，发现 use-after-free、越界、双释放等：

# 编译内核时开启
CONFIG_KASAN=y
CONFIG_KASAN_GENERIC=y

# 触发后 dmesg:
[ 5.000] ==================================================================
[ 5.000] BUG: KASAN: use-after-free in foo+0x42/0x100
[ 5.000] Read of size 4 at addr ffff888008c4f1d8 by task qemu/1234
[ 5.000] Call Trace:
[ 5.000]   foo+0x42/0x100
[ 5.000]   bar+0x10/0x20
[ 5.000] Freed by task 1233:        ←  谁释放的
[ 5.000]   kfree+...
[ 5.000] Allocated by task 1233:    ←  谁分配的
变种开销检测
KASAN2x 内存, 3x 慢UAF、越界、double-free
KFENCE几乎零开销采样式检测，可上生产
KMSAN4x 内存, 5x 慢未初始化内存读
UBSAN低未定义行为 (溢出、错位移位等)

15.6 lockdep — 在死锁发生之前就警告

# 内核配置
CONFIG_PROVE_LOCKING=y
CONFIG_DEBUG_LOCK_ALLOC=y

# 锁统计
cat /proc/lock_stat | head -30

lockdep 会跟踪每把锁的获取顺序形成有向图，发现潜在循环就报警（哪怕实际没死锁）。

15.7 bpftrace — 现代内核观察利器

# 1. 谁在打开文件?
bpftrace -e 'tracepoint:syscalls:sys_enter_openat {
    printf("%s -> %s\n", comm, str(args->filename));
}'

# 2. 哪个进程在产生 OOM?
bpftrace -e 'kprobe:oom_kill_process {
    printf("OOM kill: %s (pid %d)\n", str(arg1->name), arg1->pid);
}'

# 3. block IO 延迟分布
bpftrace -e '
tracepoint:block:block_rq_issue { @start[args->dev, args->sector] = nsecs; }
tracepoint:block:block_rq_complete /@start[args->dev, args->sector]/ {
    @us = hist((nsecs - @start[args->dev, args->sector]) / 1000);
    delete(@start[args->dev, args->sector]);
}'

# 4. TCP 连接事件
bpftrace -e 'kprobe:tcp_connect {
    printf("%s tcp_connect\n", comm);
}'

15.8 kdump + crash — 事后尸检

内核 panic 时 kdump 自动捕获完整内存到文件（vmcore），crash 工具像 gdb 一样事后分析：

# 1. 安装 + 配置 kdump (Ubuntu)
sudo apt install kdump-tools linux-crashdump
# 重启后 /var/crash 下会有 vmcore

# 2. 分析
crash /usr/lib/debug/boot/vmlinux-$(uname -r) /var/crash/.../vmcore

crash> bt          # 崩溃时的栈
crash> ps          # 当时所有进程
crash> log         # dmesg
crash> kmem -i     # 内存状况
crash> sym 0xffffffff8...   # 地址 → 符号

15.9 livepatch — 不重启打补丁

生产环境最怕"打个安全补丁要重启 1000 台机器"。livepatch 通过 ftrace 机制把函数动态替换为新版本：

/* 写一个 livepatch 模块替换 cmdline_proc_show */
#include <linux/livepatch.h>

static int my_cmdline_show(struct seq_file *m, void *v) {
    seq_puts(m, "PATCHED\n");
    return 0;
}

static struct klp_func funcs[] = {
    { .old_name = "cmdline_proc_show", .new_func = my_cmdline_show },
    { }
};
static struct klp_object objs[] = { { .funcs = funcs }, { } };
static struct klp_patch patch = { .mod = THIS_MODULE, .objs = objs };

static int __init init(void) { return klp_enable_patch(&patch); }

15.10 调试速查清单

遇到问题先问自己
  1. 能否在 QEMU 复现？ → 能则 GDB；不能则上生产
  2. 是性能问题还是正确性问题？
  3. 是 CPU/内存/IO/锁哪一类？ → 工具不同（perf / KASAN / iostat / lockstat）
  4. 是否能扛得起影响？ → 重则 ftrace 全开；轻则 bpftrace 探针
  5. 有没有崩溃 dump？ → 有则先 crash 分析
